            }

    """
    
    LABEL_CATEGORIES = {
        'steering_issues': [

            'steering_loop_detected',

            'mesh_partition',
            'excessive_hops'
        ],
        'normal_operation': [
            'optimal_performance',
            'acceptable_performance'
        ]
    }
    
    def label_event_window(self, event_window, outcome_window):
        """
        Label event window based on observed outcomes in following window
        """
        labels = {
            'issue_detected': False,
            'issue_type': None,
            'severity': None,  # critical, high, medium, low
            'resolution': None,  # auto_resolved, manual_intervention, ongoing
            'time_to_failure': None,  # minutes until issue manifested
            'resolution_time': None,  # minutes to resolve
            'SON_action_taken': None  # what WNC SON did
        }
        
        # Detect issues in outcome window
        if detect_steering_loop(outcome_window):
            labels['issue_detected'] = True
            labels['issue_type'] = 'steering_loop_detected'
            labels['severity'] = calculate_severity(outcome_window)
            labels['time_to_failure'] = calculate_ttf(event_window, outcome_window)
            labels['SON_action_taken'] = extract_son_action(outcome_window)
        
        # Similarly for other issue types...
        
        return labels
    
    def create_training_dataset(self, labeled_windows):
        """
        Create supervised learning dataset
        """
        X = []  # Features
        y = []  # Labels
        metadata = []  # Context for analysis
        
        for window in labeled_windows:
            features = extract_features(window)
            label = window.labels
            context = window.context
            
            X.append(features)
            y.append(label)
            metadata.append(context)
        
        return {
            'features': np.array(X),
            'labels': np.array(y),
            'metadata': metadata,
            'feature_names': list(features.keys()),
            'label_schema': self.LABEL_CATEGORIES
        }

            # Look back N minutes before issue manifestation
            lookback_windows = [get_preceding_events(case, minutes=30) for case in cases]
            
            # Find common patterns
            common_patterns = find_frequent_sequences(
                lookback_windows,
                min_support=0.6  # Appears in 60% of cases
            )
            
            # Calculate correlation strength
            for pattern in common_patterns:
                correlation = calculate_correlation(
                    pattern_present=pattern,
                    issue_occurred=issue_type,
                    all_windows=issue_cases
                )
                
                precursor_patterns[issue_type][pattern] = {
                    'frequency': pattern.support,
                    'correlation': correlation,
                    'avg_lead_time': calculate_avg_lead_time(pattern, issue_type),
                    'confidence': calculate_confidence(pattern, issue_type)
                }
        
        return precursor_patterns
    
    def build_causal_graph(self, correlation_data):
        """
        Build directed graph of WiFi event causality
        """
        graph = CausalGraph()
        
        # Add nodes for events and issues
        for event_type in EVENT_TYPES:
            graph.add_node(event_type, node_type='event')
        
        for issue_type in ISSUE_TYPES:
            graph.add_node(issue_type, node_type='issue')
        
        # Add edges based on correlation strength
        for event, issue, correlation in correlation_data:

                    weight=correlation.strength,

            'confidence_interval': (mean_time - 1.96*std_time, mean_time + 1.96*std_time),

class WiFiAnomalyDetector:
    """
    Autoencoder to detect unusual WiFi patterns

    def build_autoencoder(self, input_dim=50):
        """
        Build autoencoder architecture
        """
        # Encoder
        encoder = Sequential([
            Dense(32, activation='relu', input_shape=(input_dim,)),
            Dense(16, activation='relu'),
            Dense(8, activation='relu')  # Bottleneck
        ])
        
        # Decoder
        decoder = Sequential([
            Dense(16, activation='relu', input_shape=(8,)),
            Dense(32, activation='relu'),
            Dense(input_dim, activation='sigmoid')
        ])
        
        # Full autoencoder
        autoencoder = Sequential([encoder, decoder])
        autoencoder.compile(optimizer='adam', loss='mse')
        
        return autoencoder, encoder
    
    def train_on_normal_operation(self, normal_data):
        """
        Train autoencoder on WiFi data from normal operation
        """
        # Train to reconstruct normal patterns
        self.autoencoder.fit(
            normal_data,
            normal_data,  # Reconstruct input
            epochs=50,
            batch_size=32,
            validation_split=0.2,
            shuffle=True
        )
        
        # Calculate reconstruction error threshold
        reconstructions = self.autoencoder.predict(normal_data)
        reconstruction_errors = np.mean(np.square(normal_data - reconstructions), axis=1)
        
        # Set threshold at 99th percentile of normal errors
        self.anomaly_threshold = np.percentile(reconstruction_errors, 99)
    
    def detect_anomalies(self, new_data):
        """
        Detect anomalous WiFi patterns
        """
        reconstructions = self.autoencoder.predict(new_data)
        errors = np.mean(np.square(new_data - reconstructions), axis=1)
        
        anomalies = errors > self.anomaly_threshold
        
        return {
            'is_anomalous': anomalies,
            'anomaly_scores': errors,
            'severity': (errors - self.anomaly_threshold) / self.anomaly_threshold  # How far beyond threshold
        }
